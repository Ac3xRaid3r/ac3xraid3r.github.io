Win API Calls via Reflection

Resolve Address Functions

First, obtain GetModuleHandle and GetProcAddress:

## Retrieve all loaded Assemblies
$Assemblies = [AppDomain]::CurrentDomain.GetAssemblies()

## Find System.dll from GAC
$systemdll = ([AppDomain]::CurrentDomain.GetAssemblies() | Where-Object { 
  $_.GlobalAssemblyCache -And $_.Location.Split('\\')[-1].Equals('System.dll') 
})

$unsafeObj = $systemdll.GetType('Microsoft.Win32.UnsafeNativeMethods')

## Get GetModuleHandle
$GetModuleHandle = $unsafeObj.GetMethod('GetModuleHandle')

## Get GetProcAddress (handles multiple overloads)
$GetProcAddress = $unsafeObj.GetMethod('GetProcAddress',
        [reflection.bindingflags]'Public,Static', 
        $null, 
        [System.Reflection.CallingConventions]::Any,
        @([System.IntPtr], [string]), 
        $null)

Function Resolution Helper

function LookupFunc {
    Param ($moduleName, $functionName)

    $assem = ([AppDomain]::CurrentDomain.GetAssemblies() | Where-Object { 
        $_.GlobalAssemblyCache -And $_.Location.Split('\\')[-1].Equals('System.dll') 
    }).GetType('Microsoft.Win32.UnsafeNativeMethods')

    $tmp=@()
    $assem.GetMethods() | ForEach-Object {If($_.Name -eq "GetProcAddress") {$tmp+=$_}}
    return $tmp[0].Invoke($null, @(($assem.GetMethod('GetModuleHandle')).Invoke($null, @($moduleName)), $functionName))
}

Create Delegate Type

function Get-Delegate {
    Param (
        [Parameter(Position = 0, Mandatory = $True)] [IntPtr] $funcAddr,
        [Parameter(Position = 1, Mandatory = $True)] [Type[]] $argTypes,
        [Parameter(Position = 2)] [Type] $retType = [Void]
    )

    $type = [AppDomain]::CurrentDomain.DefineDynamicAssembly(
        (New-Object System.Reflection.AssemblyName('QD')), 
        [System.Reflection.Emit.AssemblyBuilderAccess]::Run
    ).DefineDynamicModule('QM', $false).DefineType(
        'QT', 'Class, Public, Sealed, AnsiClass, AutoClass', 
        [System.MulticastDelegate]
    )

    $type.DefineConstructor(
        'RTSpecialName, HideBySig, Public',
        [System.Reflection.CallingConventions]::Standard, 
        $argTypes
    ).SetImplementationFlags('Runtime, Managed')

    $type.DefineMethod(
        'Invoke', 
        'Public, HideBySig, NewSlot, Virtual', 
        $retType, 
        $argTypes
    ).SetImplementationFlags('Runtime, Managed')

    $delegate = $type.CreateType()
    return [System.Runtime.InteropServices.Marshal]::GetDelegateForFunctionPointer($funcAddr, $delegate)
}

function Get-Delegate {
    Param (
        [Parameter(Position = 0, Mandatory = $True)] [IntPtr] $funcAddr,
        [Parameter(Position = 1, Mandatory = $True)] [Type[]] $argTypes,
        [Parameter(Position = 2)] [Type] $retType = [Void]
    )

    $type = [AppDomain]::CurrentDomain.DefineDynamicAssembly(
        (New-Object System.Reflection.AssemblyName('QD')), 
        [System.Reflection.Emit.AssemblyBuilderAccess]::Run
    ).DefineDynamicModule('QM', $false).DefineType(
        'QT', 'Class, Public, Sealed, AnsiClass, AutoClass', 
        [System.MulticastDelegate]
    )
    $type.DefineConstructor('RTSpecialName, HideBySig, Public',
        [System.Reflection.CallingConventions]::Standard, $argTypes).SetImplementationFlags('Runtime, Managed')
